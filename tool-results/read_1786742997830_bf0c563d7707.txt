    48→function WeekStrip() {
    49→  const { data } = useDashboard();
    50→  const { locale, t } = useI18n();
    51→  const week = data?.weekHealth ?? [];
    52→  const today = new Date();
    53→  const days: Array<{ letter: string; date: string; logged: boolean; isToday: boolean; isFuture: boolean; dayNum: string }> = [];
    54→  for (let i = 0; i < 7; i++) {
    55→    const d = new Date(today);
    56→    d.setDate(today.getDate() - today.getDay() + i);
    57→    const key = d.toISOString().slice(0, 10);
    58→    const h = week.find((x) => x.date === key);
    59→    days.push({
    60→      letter: getWeekdayShort(d, locale),
    61→      date: key,
    62→      logged: !!(h && (h.steps > 0 || h.waterMl > 0)),
    63→      isToday: key === today.toISOString().slice(0, 10),
    64→      isFuture: d > today,
    65→      dayNum: getDayNumber(d, locale),
    66→    });
    67→  }
    68→  return (
    69→    <div className="flex items-center justify-between px-1">
    70→      {days.map((d, i) => (
    71→        <motion.div
    72→          key={i}
    73→          initial={{ opacity: 0, scale: 0.8 }}
    74→          animate={{ opacity: 1, scale: 1 }}
    75→          transition={{ delay: i * 0.04, duration: 0.3 }}
    76→          className="flex flex-col items-center gap-1.5"
    77→        >
    78→          <span className="text-[11px] font-medium text-muted-foreground">{d.letter}</span>
    79→          <div
    80→            className={cn(
    81→              "flex h-10 w-10 items-center justify-center rounded-full border-2 text-[11px] font-semibold transition-colors",
    82→              d.isToday
    83→                ? "border-foreground bg-foreground text-background"
    84→                : d.logged
    85→                ? "border-success text-success"
    86→                : d.isFuture
    87→                ? "border-muted text-muted-foreground/40"
    88→                : "border-muted text-muted-foreground"
    89→            )}
    90→          >
    91→            {d.logged && !d.isToday ? "✓" : d.dayNum}
    92→          </div>
    93→        </motion.div>
    94→      ))}
    95→    </div>
    96→  );
    97→}

   241→  const { locale, t } = useI18n();
   242→  const time = formatTime(log.timestamp, locale);
   243→
   244→  function openDetail() {

   254→    setModal(log.type === "meal" && log.mealId ? "meal-detail" : "edit-log");

   275→          <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-secondary">

   280→            <div className="flex items-center gap-3 text-xs text-muted-foreground">
   281→              <span className="flex items-center gap-1"><Flame className="h-3 w-3 text-streak" /> {formatNumber(w.caloriesBurned, locale)} {t("cal")}</span>
   282→              <span>{formatNumber(w.durationMinutes, locale)} {t("min")}</span>

   291→      <>
   292→        <div className="flex h-11 w-11 items-center justify-center overflow-hidden rounded-xl bg-secondary text-lg">

   295→          ) : (
   296→            "🍽️"
   297→          )}
   298→        </div>
   299→        <div className="flex-1 min-w-0">
   300→          <div className="truncate text-sm font-semibold">{translateFoodName(log.title, locale) ?? t("meals")}</div>
   301→          <div className="flex items-center gap-3 text-xs text-muted-foreground">
   302→            {m && (
   303→              <>
   304→                <span className="flex items-center gap-1"><Flame className="h-3 w-3 text-streak" /> {formatNumber(m.calories, locale)}</span>
   305→                <span className="flex items-center gap-1 text-protein"><Drumstick className="h-3 w-3" /> {formatNumber(Math.round(m.protein), locale)}g</span>
   306→                <span className="flex items-center gap-1 text-carbs"><Wheat className="h-3 w-3" /> {formatNumber(Math.round(m.carbs), locale)}g</span>
   307→                <span className="flex items-center gap-1 text-fats"><Droplets className="h-3 w-3" /> {formatNumber(Math.round(m.fat), locale)}g</span>
   308→              </>
   309→            )}
   310→          </div>
   311→        </div>

   316→  return (
   317→    <StaggerItem>

   500→function DateNav() {
   501→  const { selectedDate, setSelectedDate } = useApp();
   502→  const { locale, t } = useI18n();
   503→  const today = new Date().toISOString().slice(0, 10);
   504→  const isToday = selectedDate === today;
   505→  const isYesterday = selectedDate === new Date(Date.now() - 86400000).toISOString().slice(0, 10);
   506→  const dateObj = new Date(selectedDate + "T00:00:00");
   507→  const label = isToday ? t("today") : isYesterday ? t("yesterday") : formatDate(dateObj, locale, { day: true, month: "short" });
   508→
   509→  function shift(days: number) {
   510→    const d = new Date(selectedDate + "T00:00:00");
   511→    d.setDate(d.getDate() + days);
   512→    const key = d.toISOString().slice(0, 10);
   513→    if (key <= today) setSelectedDate(key);
   514→  }
   515→
   516→  return (
   517→    <div className="flex items-center justify-center gap-3">
   518→      <motion.button whileTap={{ scale: 0.9 }} onClick={() => shift(-1)} className="flex h-8 w-8 items-center justify-center rounded-full bg-secondary">
   519→        <ChevronLeft className="h-4 w-4" />
   520→      </motion.button>
   521→      <div className="min-w-24 text-center">
   522→        <div className="text-sm font-bold">{label}</div>
   523→        <div className="text-[10px] text-muted-foreground">{formatDate(dateObj, locale, { weekday: true })}</div>
   524→      </div>
   525→      <motion.button
   526→        whileTap={{ scale: 0.9 }}
   527→        onClick={() => shift(1)}
   528→        disabled={isToday}
   529→        className="flex h-8 w-8 items-center justify-center rounded-full bg-secondary disabled:opacity-30"
   530→      >
   531→        <ChevronRight className="h-4 w-4" />
   532→      </motion.button>
   533→    </div>
   534→  );
   535→}
