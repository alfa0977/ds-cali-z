    48→function WeekStrip() {
    49→  const { data } = useDashboard();
    50→  const { locale } = useI18n();
    51→  const week = data?.weekHealth ?? [];
    52→  const today = new Date();
    53→  const days: Array<{ letter: string; date: string; logged: boolean; isToday: boolean; isFuture: boolean }> = [];
    54→  for (let i = 0; i < 7; i++) {
    55→    const d = new Date(today);
    56→    d.setDate(today.getDate() - today.getDay() + i);
    57→    const key = d.toISOString().slice(0, 10);
    58→    const h = week.find((x) => x.date === key);
    59→    days.push({
    60→      letter: getWeekdayShort(d, locale),
    61→      date: key,
    62→      logged: !!(h && (h.steps > 0 || h.waterMl > 0)),
    63→      isToday: key === today.toISOString().slice(0, 10),
    64→      isFuture: d > today,
    65→    });
    66→  }
    67→  return (
    68→    <div className="flex items-center justify-between px-1">
    69→      {days.map((d, i) => (
    70→        <motion.div
    71→          key={i}
    72→          initial={{ opacity: 0, scale: 0.8 }}
    73→          animate={{ opacity: 1, scale: 1 }}
    74→          transition={{ delay: i * 0.04, duration: 0.3 }}
    75→          className="flex flex-col items-center gap-1.5"
    76→        >
    77→          <span className="text-[11px] font-medium text-muted-foreground">{d.letter}</span>
    78→          <div
    79→            className={cn(
    80→              "flex h-10 w-10 items-center justify-center rounded-full border-2 text-[11px] font-semibold transition-colors",
    81→              d.isToday
    82→                ? "border-foreground bg-foreground text-background"
    83→                : d.logged
    84→                ? "border-success text-success"
    85→                : d.isFuture
    86→                ? "border-muted text-muted-foreground/40"
    87→                : "border-muted text-muted-foreground"
    88→            )}
    89→          >
    90→            {d.logged && !d.isToday ? "✓" : getDayNumber(d.date, locale)}
    91→          </div>
    92→        </motion.div>
    93→      ))}
    94→    </div>
    95→  );
    96→}

   499→function DateNav() {
   500→  const { selectedDate, setSelectedDate } = useApp();
   501→  const { locale, t } = useI18n();
   502→  const today = new Date().toISOString().slice(0, 10);
   503→  const isToday = selectedDate === today;
   504→  const isYesterday = selectedDate === new Date(Date.now() - 86400000).toISOString().slice(0, 10);
   505→  const dateObj = new Date(selectedDate + "T00:00:00");
   506→  const label = isToday ? t("today") : isYesterday ? t("yesterday") : formatDate(dateObj, locale, { day: true, month: "short" });
   507→
   508→  function shift(days: number) {
   509→    const d = new Date(selectedDate + "T00:00:00");
   510→    d.setDate(d.getDate() + days);
   511→    const key = d.toISOString().slice(0, 10);
   512→    if (key <= today) setSelectedDate(key);
   513→  }
   514→
   515→  return (
   516→    <div className="flex items-center justify-center gap-3">
   517→      <motion.button whileTap={{ scale: 0.9 }} onClick={() => shift(-1)} className="flex h-8 w-8 items-center justify-center rounded-full bg-secondary">
   518→        <ChevronLeft className="h-4 w-4" />
   519→      </motion.button>
   520→      <div className="min-w-24 text-center">
   521→        <div className="text-sm font-bold">{label}</div>
   522→        <div className="text-[10px] text-muted-foreground">{formatDate(new Date(selectedDate + "T00:00:00"), locale, { weekday: true })}</div>
   523→      </div>
   524→      <motion.button
   525→        whileTap={{ scale: 0.9 }}
   526→        onClick={() => shift(1)}
   527→        disabled={isToday}
   528→        className="flex h-8 w-8 items-center justify-center rounded-full bg-secondary disabled:opacity-30"
   529→      >
   530→        <ChevronRight className="h-4 w-4" />
   531→      </motion.button>
   532→    </div>
   533→  );
   534→}
