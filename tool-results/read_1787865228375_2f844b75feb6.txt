   168→   168→export async function onboardUser(data: {
   169→   169→  displayName: string; sex: string; age: number; heightCm: number; weightKg: number;
   170→   170→  activityLevel: string; goal: string;
   171→   171→}): Promise<{ user: ClientUser; goals: { calories: number; protein: number; carbs: number; fat: number } }> {
   172→   172→  const bmrBase = 10 * data.weightKg + 6.25 * data.heightCm - 5 * data.age;
   173→   173→  const bmr = data.sex === "male" ? bmrBase + 5 : bmrBase - 161;
   174→   174→  const factors: Record<string, number> = { sedentary: 1.2, light: 1.375, moderate: 1.55, active: 1.725, very_active: 1.9 };
   175→   175→  let tdee = bmr * (factors[data.activityLevel] ?? 1.2);
   176→   176→  if (data.goal === "lose") tdee -= 500;
   177→   177→  else if (data.goal === "gain") tdee += 400;
   178→   178→  const goals = {
   179→   179→    calories: Math.round(tdee),
   180→   180→    protein: Math.round((tdee * 0.3) / 4),
   181→   181→    carbs: Math.round((tdee * 0.4) / 4),
   182→   182→    fat: Math.round((tdee * 0.3) / 9),
   183→   183→  };
   184→   184→  const updated = await updateUser({
   185→   185→    displayName: data.displayName, sex: data.sex, age: data.age, heightCm: data.heightCm,
   186→   186→    weightKg: data.weightKg, activityLevel: data.activityLevel, goal: data.goal,
   187→   187→    goals, onboarded: true,
   188→   188→  });
   189→   189→  return { user: updated, goals };
   190→   190→}

   194→   194→export async function searchFoods(q: string, category?: string, limit = 60): Promise<ClientFood[]> {

   198→   198→    const ql = q.trim().toLowerCase();
   199→   199→    foods = foods.filter((f: ClientFood) => f.name.toLowerCase().includes(ql));
   200→   200→  }
   201→   201→  if (category && category !== "All") {
   202→   202→    foods = foods.filter((f: ClientFood) => f.category === category);
   203→   203→  }
   204→   204→  return foods.sort((a: ClientFood, b: ClientFood) => a.name.localeCompare(b.name)).slice(0, limit);
   205→   205→}

   465→   465→// ============ DASHBOARD ============
   466→   466→
   467→   467→export async function getDashboard(date?: string): Promise<Record<string, unknown>> {
   468→   468→  await seedFoods();
   469→   469→  const db = await getDB();
   470→   470→  const user = await ensureUser();
   471→   471→  const dateParam = date ?? new Date().toISOString().slice(0, 10);
   472→   472→
   473→   473→  const allLogs = await db.getAll("logs");
   474→   474→  const userLogs = allLogs.filter((l: ClientLog) => l.userId === user.id);
   475→   475→
   476→   476→  // Day logs
   477→   477→  const dayStart = new Date(dateParam + "T00:00:00.000Z");
   478→   478→  const dayEnd = new Date(dateParam + "T23:59:59.999Z");
   479→   479→  const dayLogs = userLogs.filter((l: ClientLog) => {
   480→   480→    const ts = new Date(l.timestamp);
   481→   481→    return ts >= dayStart && ts <= dayEnd;
   482→   482→  }).sort((a: ClientLog, b: ClientLog) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   483→   483→
   484→   484→  // Calculate consumed
   485→   485→  let consumed = { calories: 0, protein: 0, carbs: 0, fat: 0 };
   486→   486→  let burned = 0;
   487→   487→  for (const log of dayLogs) {
   488→   488→    if (log.type === "meal" && log.macros) {
   489→   489→      const m = parseJSON(log.macros, { calories: 0, protein: 0, carbs: 0, fat: 0 });
   490→   490→      consumed.calories += m.calories; consumed.protein += m.protein; consumed.carbs += m.carbs; consumed.fat += m.fat;
   491→   491→    } else if (log.type === "workout" && log.workoutSummary) {
   492→   492→      const w = parseJSON(log.workoutSummary, { caloriesBurned: 0 });
   493→   493→      burned += w.caloriesBurned ?? 0;
   494→   494→    }
   495→   495→  }
   496→   496→
   497→   497→  // Today health
   498→   498→  const allHealth = await db.getAll("healthDaily");
   499→   499→  const todayHealth = allHealth.find((h: ClientHealthDaily) => h.date === dateParam) ?? null;
   500→   500→
   501→   501→  // Week health
   502→   502→  const sevenDaysAgo = new Date(dayStart);
   503→   503→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   504→   504→  const weekHealth = allHealth
   505→   505→    .filter((h: ClientHealthDaily) => h.date >= sevenDaysAgo.toISOString().slice(0, 10) && h.date <= dateParam)
   506→   506→    .sort((a: ClientHealthDaily, b: ClientHealthDaily) => a.date.localeCompare(b.date));
   507→   507→
   508→   508→  // Month health
   509→   509→  const thirtyDaysAgo = new Date(dayStart);
   510→   510→  thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 29);
   511→   511→  const monthHealth = allHealth
   512→   512→    .filter((h: ClientHealthDaily) => h.date >= thirtyDaysAgo.toISOString().slice(0, 10) && h.date <= dateParam)
   513→   513→    .sort((a: ClientHealthDaily, b: ClientHealthDaily) => a.date.localeCompare(b.date));
   514→   514→
   515→   515→  // Macro trend (7 days)
   516→   516→  const macroTrend: Array<{ date: string; calories: number; protein: number; carbs: number; fat: number }> = [];
   517→   517→  for (let i = 6; i >= 0; i--) {
   518→   518→    const d = new Date(sevenDaysAgo);
   519→   519→    d.setDate(sevenDaysAgo.getDate() + (6 - i));
   520→   520→    const key = d.toISOString().slice(0, 10);
   521→   521→    const dayLogsT = userLogs.filter((l: ClientLog) => l.type === "meal" && l.timestamp.slice(0, 10) === key);
   522→   522→    const sum = { calories: 0, protein: 0, carbs: 0, fat: 0 };
   523→   523→    for (const l of dayLogsT) {
   524→   524→      if (l.macros) {
   525→   525→        const m = parseJSON(l.macros, { calories: 0, protein: 0, carbs: 0, fat: 0 });
   526→   526→        sum.calories += m.calories; sum.protein += m.protein; sum.carbs += m.carbs; sum.fat += m.fat;
   527→   527→      }
   528→   528→    }
   529→   529→    macroTrend.push({ date: key, ...sum });
   530→   530→  }
   531→   531→
   532→   532→  // Meals by slot
   533→   533→  const mealsBySlot = {
   534→   534→    breakfast: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "breakfast" || (!l.mealSlot && new Date(l.timestamp).getHours() < 11))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   535→   535→    lunch: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "lunch" || (!l.mealSlot && new Date(l.timestamp).getHours() >= 11 && new Date(l.timestamp).getHours() < 16))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   536→   536→    dinner: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "dinner" || (!l.mealSlot && new Date(l.timestamp).getHours() >= 16))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   537→   537→    snack: dayLogs.filter((l: ClientLog) => l.type === "meal" && l.mealSlot === "snack").map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   538→   538→  };
   539→   539→
   540→   540→  // Recent logs
   541→   541→  const recentLogs = userLogs.slice(0, 20).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null }));
   542→   542→
   543→   543→  // Days logged
   544→   544→  const daysLogged = weekHealth.filter((h: ClientHealthDaily) => h.steps > 0 || h.waterMl > 0).length;
   545→   545→
   546→   546→  // Streak
   547→   547→  const loggedDays = new Set(userLogs.filter((l: ClientLog) => l.type === "meal").map((l: ClientLog) => l.timestamp.slice(0, 10)));
   548→   548→  let streak = 0;
   549→   549→  for (let i = 0; i < 365; i++) {
   550→   550→    const d = new Date(dayStart);
   551→   551→    d.setDate(d.getDate() - i);
   552→   552→    if (loggedDays.has(d.toISOString().slice(0, 10))) streak++;
   553→   553→    else if (i > 0) break;
   554→   554→  }
   555→   555→
   556→   556→  return {
   557→   557→    user: { ...user, goals: user.goals, streak },
   558→   558→    todayHealth: todayHealth ? { ...todayHealth, workouts: parseJSON(todayHealth.workouts, []) } : null,
   559→   559→    consumed, burned,
   560→   560→    netCalories: consumed.calories - burned,
   561→   561→    dayLogs: dayLogs.map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   562→   562→    mealsBySlot,
   563→   563→    weekHealth, monthHealth, macroTrend,
   564→   564→    recentLogs,
   565→   565→    daysLogged,
   566→   566→  };
   567→   567→}

   571→   571→export async function getMealSuggestions(slot?: string): Promise<{ suggestions: ClientFood[]; remaining: { calories: number; protein: number; carbs: number; fat: number }; biggestGap: string }> {
   572→   572→  const dash = await getDashboard();
   573→   573→  const goals = (dash.user as ClientUser).goals;
   574→   574→  const consumed = dash.consumed as { calories: number; protein: number; carbs: number; fat: number };
   575→   575→  const remaining = {
   576→   576→    calories: Math.max(0, goals.calories - consumed.calories),
   577→   577→    protein: Math.max(0, goals.protein - consumed.protein),
   578→   578→    carbs: Math.max(0, goals.carbs - consumed.carbs),
   579→   579→    fat: Math.max(0, goals.fat - consumed.fat),
   580→   580→  };
   581→   581→  const proteinPct = goals.protein > 0 ? remaining.protein / goals.protein : 0;
   582→   582→  const carbsPct = goals.carbs > 0 ? remaining.carbs / goals.carbs : 0;
   583→   583→  const fatPct = goals.fat > 0 ? remaining.fat / goals.fat : 0;
   584→   584→  let biggestGap = "protein";
   585→   585→  if (proteinPct >= carbsPct && proteinPct >= fatPct) biggestGap = "protein";
   586→   586→  else if (carbsPct >= fatPct) biggestGap = "carbs";
   587→   587→  else biggestGap = "fat";
   588→   588→  const foods = await searchFoods("", undefined, 30);
   589→   589→  const sorted = foods.sort((a: ClientFood, b: ClientFood) => (b[biggestGap as "protein"] as number) - (a[biggestGap as "protein"] as number));
   590→   590→  return { suggestions: sorted.slice(0, 8), remaining, biggestGap };
   591→   591→}

   619→   619→export async function lookupBarcode(code: string): Promise<{ food: ClientFood; source: string }> {
   620→   620→  const db = await getDB();
   621→   621→  const allFoods = await db.getAll("foods");
   622→   622→  const local = allFoods.find((f: ClientFood) => f.barcode === code);
   623→   623→  if (local) return { food: local, source: "local" };
   624→   624→  // Try Open Food Facts
   625→   625→  const res = await fetch(`https://world.openfoodfacts.org/api/v2/product/${encodeURIComponent(code)}.json`, {
   626→   626→    headers: { "User-Agent": "DS-Cali/1.0" },
   627→   627→  });
   628→   628→  const data = await res.json();
   629→   629→  if (data.status !== 1 || !data.product) throw new Error("Product not found");
   630→   630→  const p = data.product;
   631→   631→  const n = p.nutriments || {};
   632→   632→  const servingG = p.serving_quantity ? Number(p.serving_quantity) : 100;
   633→   633→  const ratio = servingG / 100;
   634→   634→  const food: ClientFood = {
   635→   635→    id: genId(),
   636→   636→    name: p.product_name || "Unknown product",
   637→   637→    servingSize: servingG ? `${servingG}g` : "100g",
   638→   638→    servingWeightGrams: servingG || 100,
   639→   639→    calories: Math.round((Number(n["energy-kcal_100g"]) || Number(n.energy_100g) / 4.184 || 0) * ratio),
   640→   640→    protein: Math.round((Number(n.proteins_100g) || 0) * ratio * 10) / 10,
   641→   641→    carbs: Math.round((Number(n.carbohydrates_100g) || 0) * ratio * 10) / 10,
   642→   642→    fat: Math.round((Number(n.fat_100g) || 0) * ratio * 10) / 10,
   643→   643→    source: "database", barcode: code,
   644→   644→    category: p.categories_tags?.[0]?.replace("en:", "") ?? null,
   645→   645→    emoji: "📦", densityGramsPerMl: null, createdBy: "demo",
   646→   646→    createdAt: new Date().toISOString(),
   647→   647→  };
   648→   648→  await db.put("foods", food);
   649→   649→  return { food, source: "openfoodfacts" };
   650→   650→}
