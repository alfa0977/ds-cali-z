    99→let dbInstance: IDBPDatabase | null = null;

   101→async function getDB(): Promise<IDBPDatabase> {
   102→  if (dbInstance) return dbInstance;
   103→  dbInstance = await openDB(DB_NAME, DB_VERSION, {
   104→    upgrade(db) {
   105→      if (!db.objectStoreNames.contains("users")) db.createObjectStore("users", { keyPath: "id" });
   106→      if (!db.objectStoreNames.contains("foods")) db.createObjectStore("foods", { keyPath: "id" });
   107→      if (!db.objectStoreNames.contains("meals")) db.createObjectStore("meals", { keyPath: "id" });
   108→      if (!db.objectStoreNames.contains("logs")) db.createObjectStore("logs", { keyPath: "id" });
   109→      if (!db.objectStoreNames.contains("healthDaily")) db.createObjectStore("healthDaily", { keyPath: "id" });
   110→      if (!db.objectStoreNames.contains("favorites")) db.createObjectStore("favorites", { keyPath: "id" });
   111→    },
   112→  });
   113→  return dbInstance;
   114→}

   120→function parseJSON<T>(s: string | null | undefined, fallback: T): T {
   121→  if (!s) return fallback;
   122→  try { return JSON.parse(s) as T; } catch { return fallback; }
   123→}

   168→export async function onboardUser(data: {
   169→  displayName: string; sex: string; age: number; heightCm: number; weightKg: number;
   170→  activityLevel: string; goal: string;
   171→}): Promise<{ user: ClientUser; goals: { calories: number; protein: number; carbs: number; fat: number } }> {
   172→  const bmrBase = 10 * data.weightKg + 6.25 * data.heightCm - 5 * data.age;
   173→  const bmr = data.sex === "male" ? bmrBase + 5 : bmrBase - 161;
   174→  const factors: Record<string, number> = { sedentary: 1.2, light: 1.375, moderate: 1.55, active: 1.725, very_active: 1.9 };
   175→  let tdee = bmr * (factors[data.activityLevel] ?? 1.2);
   176→  if (data.goal === "lose") tdee -= 500;
   177→  else if (data.goal === "gain") tdee += 400;
   178→  const goals = {
   179→    calories: Math.round(tdee),
   180→    protein: Math.round((tdee * 0.3) / 4),
   181→    carbs: Math.round((tdee * 0.4) / 4),
   182→    fat: Math.round((tdee * 0.3) / 9),
   183→  };
   184→  const updated = await updateUser({
   185→    displayName: data.displayName, sex: data.sex, age: data.age, heightCm: data.heightCm,
   186→    weightKg: data.weightKg, activityLevel: data.activityLevel, goal: data.goal,
   187→    goals, onboarded: true,
   188→  });
   189→  return { user: updated, goals };
   190→}

   194→export async function searchFoods(q: string, category?: string, limit = 60): Promise<ClientFood[]> {
   195→  const db = await getDB();
   196→  let foods = await db.getAll("foods");
   197→  if (q.trim()) {
   198→    const ql = q.trim().toLowerCase();
   199→    foods = foods.filter((f: ClientFood) => f.name.toLowerCase().includes(ql));
   200→  }
   201→  if (category && category !== "All") {
   202→    foods = foods.filter((f: ClientFood) => f.category === category);
   203→  }
   204→  return foods.sort((a: ClientFood, b: ClientFood) => a.name.localeCompare(b.name)).slice(0, limit);
   205→}

   352→export async function logFood(foodId: string, servings: number, mealSlot?: string): Promise<{ mealId: string; logId: string }> {

   358→    protein: Math.round(food.protein * servings * 10) / 10,
   359→    carbs: Math.round(food.carbs * servings * 10) / 10,
   360→    fat: Math.round(food.fat * servings * 10) / 10,
   361→  };
   362→  return logMeal({
   363→    source: "manual", ingredients: [{ name: food.name, estimatedWeightGrams: Math.round(food.servingWeightGrams * servings), confidence: 1 }],
   364→    macros, healthScore: 60, title: food.name, mealSlot,
   365→  });

   368→export async function logWater(deltaMl: number, date?: string): Promise<{ waterMl: number }> {
   369→  const db = await getDB();
   370→  const userId = await getActiveUserId();
   371→  const dateKey = date ?? new Date().toISOString().slice(0, 10);
   372→  const existing = await db.getAll("healthDaily");
   373→  const dayHealth = existing.find((h: ClientHealthDaily) => h.date === dateKey);
   374→  const current = dayHealth?.waterMl ?? 0;
   375→  const next = Math.max(0, current + deltaMl);
   376→  if (dayHealth) {
   377→    await db.put("healthDaily", { ...dayHealth, waterMl: next, updatedAt: new Date().toISOString() });
   378→  } else {
   379→    await db.put("healthDaily", { id: genId(), userId, date: dateKey, waterMl: next, steps: 0, activeEnergyKcal: 0, weightKg: null, workouts: "[]", updatedAt: new Date().toISOString() });
   380→  }
   381→  await db.put("logs", { id: genId(), userId, type: "water", waterMl: deltaMl, timestamp: new Date().toISOString(), mealId: null, macros: null, workoutSummary: null, imageUrl: null, title: null, mealSlot: null, corrected: false });
   382→  return { waterMl: next };
   383→}

   409→export async function updateLog(data: { logId: string; title?: string; macros?: { calories: number; protein: number; carbs: number; fat: number }; mealSlot?: string; ingredients?: Array<Record<string, unknown>> }): Promise<void> {

   414→  if (data.title) logData.title = data.title;
   415→  if (data.macros) logData.macros = JSON.stringify(data.macros);
   416→  if (data.mealSlot) logData.mealSlot = data.mealSlot;
   417→  await db.put("logs", { ...log, ...logData });
   418→  if (log.mealId) {
   419→    const meal = await db.get("meals", log.mealId);
   420→    if (meal) {
   421→      const mealData: Partial<ClientMeal> = {};
   422→      if (data.title) mealData.title = data.title;
   423→      if (data.macros) mealData.macros = JSON.stringify(data.macros);
   424→      if (data.ingredients) mealData.ingredients = JSON.stringify(data.ingredients);
   425→      await db.put("meals", { ...meal, ...mealData });
   426→    }
   427→  }
   428→}

   467→export async function getDashboard(date?: string): Promise<Record<string, unknown>> {
   468→  await seedFoods();
   469→  const db = await getDB();
   470→  const user = await ensureUser();
   471→  const dateParam = date ?? new Date().toISOString().slice(0, 10);
   472→
   473→  const allLogs = await db.getAll("logs");
   474→  const userLogs = allLogs.filter((l: ClientLog) => l.userId === user.id);
   475→
   476→  // Day logs
   477→  const dayStart = new Date(dateParam + "T00:00:00.000Z");
   478→  const dayEnd = new Date(dateParam + "T23:59:59.999Z");
   479→  const dayLogs = userLogs.filter((l: ClientLog) => {
   480→    const ts = new Date(l.timestamp);
   481→    return ts >= dayStart && ts <= dayEnd;
   482→  }).sort((a: ClientLog, b: ClientLog) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   483→
   484→  // Calculate consumed
   485→  let consumed = { calories: 0, protein: 0, carbs: 0, fat: 0 };
   486→  let burned = 0;
   487→  for (const log of dayLogs) {
   488→    if (log.type === "meal" && log.macros) {
   489→      const m = parseJSON(log.macros, { calories: 0, protein: 0, carbs: 0, fat: 0 });
   490→      consumed.calories += m.calories; consumed.protein += m.protein; consumed.carbs += m.carbs; consumed.fat += m.fat;
   491→    } else if (log.type === "workout" && log.workoutSummary) {
   492→      const w = parseJSON(log.workoutSummary, { caloriesBurned: 0 });
   493→      burned += w.caloriesBurned ?? 0;
   494→    }
   495→  }
   496→
   497→  // Today health
   498→  const allHealth = await db.getAll("healthDaily");
   499→  const todayHealth = allHealth.find((h: ClientHealthDaily) => h.date === dateParam) ?? null;
   500→
   501→  // Week health
   502→  const sevenDaysAgo = new Date(dayStart);
   503→  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 6);
   504→  const weekHealth = allHealth
   505→    .filter((h: ClientHealthDaily) => h.date >= sevenDaysAgo.toISOString().slice(0, 10) && h.date <= dateParam)
   506→    .sort((a: ClientHealthDaily, b: ClientHealthDaily) => a.date.localeCompare(b.date));
   507→
   508→  // Month health
   509→  const thirtyDaysAgo = new Date(dayStart);
   510→  thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 29);
   511→  const monthHealth = allHealth
   512→    .filter((h: ClientHealthDaily) => h.date >= thirtyDaysAgo.toISOString().slice(0, 10) && h.date <= dateParam)
   513→    .sort((a: ClientHealthDaily, b: ClientHealthDaily) => a.date.localeCompare(b.date));
   514→
   515→  // Macro trend (7 days)
   516→  const macroTrend: Array<{ date: string; calories: number; protein: number; carbs: number; fat: number }> = [];
   517→  for (let i = 6; i >= 0; i--) {
   518→    const d = new Date(sevenDaysAgo);
   519→    d.setDate(sevenDaysAgo.getDate() + (6 - i));
   520→    const key = d.toISOString().slice(0, 10);
   521→    const dayLogsT = userLogs.filter((l: ClientLog) => l.type === "meal" && l.timestamp.slice(0, 10) === key);
   522→    const sum = { calories: 0, protein: 0, carbs: 0, fat: 0 };
   523→    for (const l of dayLogsT) {
   524→      if (l.macros) {
   525→        const m = parseJSON(l.macros, { calories: 0, protein: 0, carbs: 0, fat: 0 });
   526→        sum.calories += m.calories; sum.protein += m.protein; sum.carbs += m.carbs; sum.fat += m.fat;
   527→      }
   528→    }
   529→    macroTrend.push({ date: key, ...sum });
   530→  }
   531→
   532→  // Meals by slot
   533→  const mealsBySlot = {
   534→    breakfast: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "breakfast" || (!l.mealSlot && new Date(l.timestamp).getHours() < 11))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   535→    lunch: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "lunch" || (!l.mealSlot && new Date(l.timestamp).getHours() >= 11 && new Date(l.timestamp).getHours() < 16))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   536→    dinner: dayLogs.filter((l: ClientLog) => l.type === "meal" && (l.mealSlot === "dinner" || (!l.mealSlot && new Date(l.timestamp).getHours() >= 16))).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   537→    snack: dayLogs.filter((l: ClientLog) => l.type === "meal" && l.mealSlot === "snack").map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   538→  };
   539→
   540→  // Recent logs
   541→  const recentLogs = userLogs.slice(0, 20).map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null }));
   542→
   543→  // Days logged
   544→  const daysLogged = weekHealth.filter((h: ClientHealthDaily) => h.steps > 0 || h.waterMl > 0).length;
   545→
   546→  // Streak
   547→  const loggedDays = new Set(userLogs.filter((l: ClientLog) => l.type === "meal").map((l: ClientLog) => l.timestamp.slice(0, 10)));
   548→  let streak = 0;
   549→  for (let i = 0; i < 365; i++) {
   550→    const d = new Date(dayStart);
   551→    d.setDate(d.getDate() - i);
   552→    if (loggedDays.has(d.toISOString().slice(0, 10))) streak++;
   553→    else if (i > 0) break;
   554→  }
   555→
   556→  return {
   557→    user: { ...user, goals: user.goals, streak },
   558→    todayHealth: todayHealth ? { ...todayHealth, workouts: parseJSON(todayHealth.workouts, []) } : null,
   559→    consumed, burned,
   560→    netCalories: consumed.calories - burned,
   561→    dayLogs: dayLogs.map((l: ClientLog) => ({ ...l, macros: l.macros ? parseJSON(l.macros, null) : null, workoutSummary: l.workoutSummary ? parseJSON(l.workoutSummary, null) : null })),
   562→    mealsBySlot,
   563→    weekHealth, monthHealth, macroTrend,
   564→    recentLogs,
   565→    daysLogged,
   566→  };
   567→}

   571→export async function getMealSuggestions(slot?: string): Promise<{ suggestions: ClientFood[]; remaining: { calories: number; protein: number; carbs: number; fat: number }; biggestGap: string }> {
   572→  const dash = await getDashboard();
   573→  const goals = (dash.user as ClientUser).goals;
   574→  const consumed = dash.consumed as { calories: number; protein: number; carbs: number; fat: number };
   575→  const remaining = {
   576→    calories: Math.max(0, goals.calories - consumed.calories),
   577→    protein: Math.max(0, goals.protein - consumed.protein),
   578→    carbs: Math.max(0, goals.carbs - consumed.carbs),
   579→    fat: Math.max(0, goals.fat - consumed.fat),
   580→  };
   581→  const proteinPct = goals.protein > 0 ? remaining.protein / goals.protein : 0;
   582→  const carbsPct = goals.carbs > 0 ? remaining.carbs / goals.carbs : 0;
   583→  const fatPct = goals.fat > 0 ? remaining.fat / goals.fat : 0;
   584→  let biggestGap = "protein";
   585→  if (proteinPct >= carbsPct && proteinPct >= fatPct) biggestGap = "protein";
   586→  else if (carbsPct >= fatPct) biggestGap = "carbs";
   587→  else biggestGap = "fat";
   588→  const foods = await searchFoods("", undefined, 30);
   589→  const sorted = foods.sort((a: ClientFood, b: ClientFood) => (b[biggestGap as "protein"] as number) - (a[biggestGap as "protein"] as number));
   590→  return { suggestions: sorted.slice(0, 8), remaining, biggestGap };
   591→}

   619→export async function lookupBarcode(code: string): Promise<{ food: ClientFood; source: string }> {
   620→  const db = await getDB();
   621→  const allFoods = await db.getAll("foods");
   622→  const local = allFoods.find((f: ClientFood) => f.barcode === code);
   623→  if (local) return { food: local, source: "local" };
   624→  // Try Open Food Facts
   625→  const res = await fetch(`https://world.openfoodfacts.org/api/v2/product/${encodeURIComponent(code)}.json`, {
   626→    headers: { "User-Agent": "DS-Cali/1.0" },
   627→  });
   628→  const data = await res.json();
   629→  if (data.status !== 1 || !data.product) throw new Error("Product not found");
   630→  const p = data.product;
   631→  const n = p.nutriments || {};
   632→  const servingG = p.serving_quantity ? Number(p.serving_quantity) : 100;
   633→  const ratio = servingG / 100;
   634→  const food: ClientFood = {
   635→    id: genId(),
   636→    name: p.product_name || "Unknown product",
   637→    servingSize: servingG ? `${servingG}g` : "100g",
   638→    servingWeightGrams: servingG || 100,
   639→    calories: Math.round((Number(n["energy-kcal_100g"]) || Number(n.energy_100g) / 4.184 || 0) * ratio),
   640→    protein: Math.round((Number(n.proteins_100g) || 0) * ratio * 10) / 10,
   641→    carbs: Math.round((Number(n.carbohydrates_100g) || 0) * ratio * 10) / 10,
   642→    fat: Math.round((Number(n.fat_100g) || 0) * ratio * 10) / 10,
   643→    source: "database", barcode: code,
   644→    category: p.categories_tags?.[0]?.replace("en:", "") ?? null,
   645→    emoji: "📦", densityGramsPerMl: null, createdBy: "demo",
   646→    createdAt: new Date().toISOString(),
   647→  };
   648→  await db.put("foods", food);
   649→  return { food, source: "openfoodfacts" };
   650→}
