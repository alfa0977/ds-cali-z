   507→  needMoreFats: { fa: "نیاز به چربی بیشتر", en: "Need more fats" },
   508→} as const;
   509→
   510→export type TranslationKey = keyof typeof translations;
   511→
   512→interface I18nContext {
   513→  locale: Locale;
   514→  setLocale: (l: Locale) => void;
   515→  t: (key: TranslationKey) => string;
   516→  dir: "rtl" | "ltr";
   517→}
   518→
   519→const I18nContext = createContext<I18nContext | null>(null);
   520→
   521→const STORAGE_KEY = "ds-cali-locale";
   522→
   523→export function I18nProvider({ children }: { children: ReactNode }) {
   524→  // Default to Persian (fa). Read from localStorage on mount.
   525→  const [locale, setLocaleState] = useState<Locale>("fa");
   526→
   527→  useEffect(() => {
   528→    const timer = setTimeout(() => {
   529→      try {
   530→        const saved = localStorage.getItem(STORAGE_KEY) as Locale | null;
   531→        if (saved === "fa" || saved === "en") {
   532→          setLocaleState(saved);
   533→        }
   534→      } catch {}
   535→    }, 0);
   536→    return () => clearTimeout(timer);
   537→  }, []);
   538→
   539→  useEffect(() => {
   540→    try {
   541→      localStorage.setItem(STORAGE_KEY, locale);
   542→    } catch {}
   543→    document.documentElement.lang = locale;
   544→    document.documentElement.dir = locale === "fa" ? "rtl" : "ltr";
   545→  }, [locale]);
   546→
   547→  const setLocale = (l: Locale) => setLocaleState(l);
   548→  const t = (key: TranslationKey) => translations[key]?.[locale] ?? String(key);
   549→  const dir = locale === "fa" ? "rtl" : "ltr";
   550→
   551→  return (
   552→    <I18nContext.Provider value={{ locale, setLocale, t, dir }}>
   553→      {children}
   554→    </I18nContext.Provider>
   555→  );
   556→}
   557→
   558→export function useI18n() {
   559→  const ctx = useContext(I18nContext);
   560→  if (!ctx) {
   561→    // Fallback for SSR / pre-provider
   562→    return {
   563→      locale: "fa" as Locale,
   564→      setLocale: () => {},
   565→      t: (key: TranslationKey) => translations[key]?.fa ?? String(key),
   566→      dir: "rtl" as const,
   567→    };
   568→  }
   569→  return ctx;
   570→}
